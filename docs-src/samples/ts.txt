@@@ conditional
const first = await client.read(url);

// Revalidate: 304 Not Modified is a normal result, not an error.
const again = await client.read(url, { ifNoneMatch: first.etag! });
if (again.notModified) console.log("cached copy is still fresh");

// Optimistic concurrency: only overwrite the version you read.
try {
  await client.update(url, "new content", "text/plain", { ifMatch: first.etag! });
} catch (e) {
  if (!(e instanceof PreconditionFailedError)) throw e;
  // Someone else changed it (412): re-read, merge, retry.
}

@@@ patch-linkset
const doc = await client.readLinkset(url);
const rel = "https://example.org/rel/reviewer";           // extension relation types are URIs
const reviewer = { href: "https://id.example/bob" };

// JsonPointer escapes "/" in the relation URI as "~1".
// "add" on an existing member replaces it, so append with "/-" when the relation exists.
const patch = doc.linkset.targets(rel).length > 0
  ? new JsonPatch().add(JsonPointer.from(["linkset", 0, rel, "-"]), reviewer)
  : new JsonPatch().add(JsonPointer.from(["linkset", 0, rel]), [reviewer]);

await client.patchLinkset(doc.url, patch, { ifMatch: doc.etag! });

@@@ custom-authenticator
// Any scheme (API keys, cookies, DPoP…) plugs in through the Authenticator interface.
class ApiKeyAuthenticator implements Authenticator {
  readonly #key: string;
  constructor(key: string) {
    this.#key = key;
  }
  authorize(request: AuthRequest): void {
    if (new URL(request.url).origin === "https://storage.example") {
      request.headers.set("x-api-key", this.#key);
    }
  }
  handleChallenge(): boolean {
    return false;   // nothing to refresh: surface the 401 as UnauthorizedError
  }
}
const client = new LwsClient({ authenticator: new ApiKeyAuthenticator(apiKey) });

@@@ errors
try {
  await client.read(url);
} catch (e) {
  if (e instanceof NotFoundError) {
    console.log("no such resource");
  } else if (e instanceof UnauthorizedError) {
    console.log("not authenticated", e.challenges);
  } else if (e instanceof HttpError) {
    console.log(e.status, e.problem?.title ?? e.message);
  } else if (e instanceof AuthenticationError) {
    console.log("token exchange failed:", e.error, e.errorDescription);
  } else {
    throw e;
  }
}
