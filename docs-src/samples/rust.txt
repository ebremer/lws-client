use lws_client::{Client, SelfSignedCredentials, TokenExchangeAuthenticator, Url, crypto::SigningKey};

#[tokio::main]
async fn main() -> lws_client::Result<()> {
    let credentials = SelfSignedCredentials::did_key(SigningKey::generate_p256()?);
    println!("agent:   {}", credentials.agent());
    let client = Client::builder()
        .authenticator(TokenExchangeAuthenticator::new(credentials))
        .build()?;

    let storage = client.discover_storage("http://localhost:8787/root/").await?;
    let root = storage.storage_root()?.clone();
    println!("storage: {}  root: {root}", storage.id);

    let folder = client.create_container(&root).slug("hello-world").await?.location;
    let file = client
        .create(&folder, "Hello, LWS!", "text/plain")
        .slug("hello.txt")
        .await?
        .location;
    println!("created: {file}");

    let resource = client.read(&file).await?;
    let etag = resource.metadata.etag.as_deref().unwrap_or("-");
    let parent = resource.metadata.parent().map(Url::as_str).unwrap_or("-");
    println!("{}  (etag {etag}, parent {parent})", resource.text()?);
    Ok(())
}

use lws_client::auth::{Authenticator, BoxFuture, RequestParts, ResponseParts};

// Any scheme (API keys, cookies, DPoP…) plugs in through the Authenticator trait.
struct ApiKeyAuthenticator {
    key: String,
}

impl Authenticator for ApiKeyAuthenticator {
    fn authorize<'a>(&'a self, request: &'a mut RequestParts) -> BoxFuture<'a, lws_client::Result<()>> {
        Box::pin(async move {
            if request.url.host_str() == Some("storage.example") {
                request.headers.insert("x-api-key", self.key.parse().expect("valid header value"));
            }
            Ok(())
        })
    }

    fn handle_challenge<'a>(
        &'a self,
        _request: &'a RequestParts,
        _response: &'a ResponseParts,
    ) -> BoxFuture<'a, lws_client::Result<bool>> {
        Box::pin(async { Ok(false) }) // nothing to refresh: the 401 surfaces as Error::Unauthorized
    }
}

fn api_key_client(key: &str) -> lws_client::Result<Client> {
    Client::builder()
        .authenticator(ApiKeyAuthenticator { key: key.to_owned() })
        .build()
}

use lws_client::{Client, Result, Url, WebhookVerifier};

/// Builds a verifier that only accepts deliveries signed by one storage.
fn verifier() -> WebhookVerifier {
    let storage = Url::parse("https://storage.example/").expect("valid URL");
    WebhookVerifier::builder()
        .client(Client::new())        // fetches storage descriptions (and their keys)
        .trusted_storage(storage)     // only this storage's signatures
        .build()
}

/// Call this from your web framework's handler with the raw request; returns the status to send.
async fn handle(verifier: &WebhookVerifier, request: http::Request<Vec<u8>>) -> Result<u16> {
    // The URL you registered (your public URL when behind a proxy), not the Host header.
    let inbox = Url::parse("https://my-app.example/hooks/lws").expect("valid URL");
    match verifier.verify_request(&inbox, &request).await {
        Ok(verified) => {
            for a in &verified.notification.activities {
                println!("{:?} {} {:?}", a.types, a.object.id, a.published);
            }
            Ok(204)
        }
        Err(e) if e.is_signature_verification() => Ok(401),
        Err(e) => Err(e),
    }
}
