	"context"

	"strings"

	lws "github.com/ebremer/lws-client/go"

func main() {
	ctx := context.Background()
	key, err := lws.GenerateP256Key()
	if err != nil {
		log.Fatal(err)
	}
	credentials, err := lws.NewDIDKeyCredentials(key, nil)
	if err != nil {
		log.Fatal(err)
	}
	client := lws.NewClient(lws.WithAuthenticator(lws.NewTokenExchangeAuthenticator(credentials, nil)))
	fmt.Println("agent:  ", credentials.Agent())

	storage, err := client.DiscoverStorage(ctx, "http://localhost:8787/root/")
	if err != nil {
		log.Fatal(err)
	}
	root, err := storage.StorageRoot()
	if err != nil {
		log.Fatal(err)
	}
	fmt.Println("storage:", storage.ID, " root:", root)

	folder, err := client.CreateContainer(ctx, root, lws.Slug("hello-world"))
	if err != nil {
		log.Fatal(err)
	}
	file, err := client.Create(ctx, folder.Location, strings.NewReader("Hello, LWS!"), "text/plain", lws.Slug("hello.txt"))
	if err != nil {
		log.Fatal(err)
	}
	fmt.Println("created:", file.Location)

	res, err := client.Read(ctx, file.Location)
	if err != nil {
		log.Fatal(err)
	}
	fmt.Printf("%s  (etag %s, parent %s)\n", res.Text(), res.ETag, res.Parent)
}

// Any scheme (API keys, cookies, DPoP…) plugs in through the Authenticator interface.
type apiKeyAuthenticator struct{ key string }

func (a apiKeyAuthenticator) Authorize(ctx context.Context, req *http.Request) error {
	if req.URL.Host == "storage.example" {
		req.Header.Set("X-Api-Key", a.key)
	}
	return nil
}

func (a apiKeyAuthenticator) HandleChallenge(ctx context.Context, req *http.Request, resp *http.Response) (bool, error) {
	return false, nil // nothing to refresh: the 401 surfaces as an *lws.HTTPError
}

func newAPIKeyClient(apiKey string) *lws.Client {
	return lws.NewClient(lws.WithAuthenticator(apiKeyAuthenticator{key: apiKey}))
}

package main

import (
	"context"
	"fmt"
	"log"
	"net/http"

	lws "github.com/ebremer/lws-client/go"
)

func main() {
	verifier := lws.NewWebhookVerifier(&lws.WebhookVerifierOptions{
		TrustedStorages: []string{"https://storage.example/"}, // only this storage's signatures
	})
	inbox := "https://my-app.example/hooks/lws" // the registered URL (your public URL behind a proxy)

	// Handler answers 204 for verified deliveries and 401 otherwise.
	http.Handle("/hooks/lws", verifier.Handler(inbox, func(ctx context.Context, n *lws.VerifiedNotification) {
		for _, a := range n.Notification.Activities {
			fmt.Println(a.Types, a.Object.ID, a.Published)
		}
	}))
	log.Fatal(http.ListenAndServe(":8080", nil))

	// In your own handler: verified, err := verifier.VerifyRequest(r, inbox)
}

var authErr *lws.AuthenticationError

case err == nil:
case errors.Is(err, lws.ErrNotFound):
	fmt.Println("no such resource")
case errors.Is(err, lws.ErrUnauthorized) && errors.As(err, &httpErr):
	fmt.Println("not authenticated", httpErr.Challenges)
case errors.As(err, &httpErr):
	fmt.Println(httpErr.StatusCode, httpErr.Problem)
case errors.As(err, &authErr):
	fmt.Println("token exchange failed:", authErr.Code, authErr.Description)
default:
	return err
}
