fun main() = runBlocking {
    val credentials = SelfSignedCredentials.didKey(SigningKey.generateP256())
    val client = LwsClient(authenticator = TokenExchangeAuthenticator(credentials))
    println("agent:   ${credentials.agent}")

    val storage = client.discoverStorage(URI("http://localhost:8787/root/"))
    println("storage: ${storage.id}  root: ${storage.storageRoot()}")

    val folder = client.createContainer(storage.storageRoot(), slug = "hello-world").location
    val file = client.createText(folder, "Hello, LWS!", slug = "hello.txt").location
    println("created: $file")

    val resource = client.read(file)
    println("${resource.text()}  (etag ${resource.etag ?: "-"}, parent ${resource.metadata.parent ?: "-"})")
}

val reviewer = buildJsonObject { put("href", "https://id.example/bob") }

// JsonPointer escapes "/" in the relation URI as "~1".
// "add" on an existing member replaces it, so append with "/-" when the relation exists.
val patch = if (doc.linkset.targets(rel).isEmpty()) {
    jsonPatch { add(JsonPointer.fromSegments("linkset", 0, rel), JsonArray(listOf(reviewer))) }
} else {
    jsonPatch { add(JsonPointer.fromSegments("linkset", 0, rel, "-"), reviewer) }
}
client.patchLinkset(doc.url, patch, ifMatch = doc.etag)

import com.ebremer.lws.kotlin.LwsClient
import com.ebremer.lws.kotlin.auth.AuthRequest
import com.ebremer.lws.kotlin.auth.AuthResponse
import com.ebremer.lws.kotlin.auth.Authenticator

// Any scheme (API keys, cookies, DPoP…) plugs in through the Authenticator interface.
class ApiKeyAuthenticator(private val key: String) : Authenticator {
    override suspend fun authorize(request: AuthRequest): AuthRequest =
        if (request.url.host == "storage.example") request.withHeaders(request.headers.with("X-Api-Key", key)) else request

    // Nothing to refresh: the 401 surfaces as UnauthorizedException.
    override suspend fun handleChallenge(request: AuthRequest, response: AuthResponse): Boolean = false

    override fun toString(): String = "ApiKeyAuthenticator"   // keeps the key out of logs
}

val client = LwsClient(authenticator = ApiKeyAuthenticator("my-api-key"))

@@@ webhook
import com.ebremer.lws.kotlin.LwsClient
import com.ebremer.lws.kotlin.SignatureVerificationException
import com.ebremer.lws.kotlin.notify.WebhookVerifier
import com.sun.net.httpserver.HttpServer
import kotlinx.coroutines.runBlocking
import java.net.InetSocketAddress
import java.net.URI

fun main() {
    val verifier = WebhookVerifier(
        client = LwsClient(),                                       // fetches storage descriptions
        trustedStorages = listOf(URI("https://storage.example/")),  // only this storage's keys
    )
    val inbox = URI("https://my-app.example/hooks/lws")             // the registered (public) URL

    val server = HttpServer.create(InetSocketAddress(8080), 0)
    server.createContext("/hooks/lws") { exchange ->
        val status = runBlocking {
            try {
                val verified = verifier.verifyExchange(exchange, inbox)
                for (a in verified.notification.activities) println("${a.types} ${a.`object`.id} ${a.published}")
                204
            } catch (e: SignatureVerificationException) {
                System.err.println("rejected: ${e.message}")
                401
            }
        }
        exchange.sendResponseHeaders(status, -1)
        exchange.close()
    }
    server.start()
    // Ktor, Spring, …: verifier.verify(method, inbox, headers, body) with the request's parts
}

@@@ errors
try {
    client.read(url)
} catch (_: NotFoundException) {
    println("no such resource")
} catch (e: UnauthorizedException) {
    println("not authenticated: ${e.challenges.size} challenge(s)")
} catch (e: HttpException) {                           // any other error status
    println("${e.status} ${e.problem?.title ?: e.message}")
} catch (e: AuthenticationException) {
    println("token exchange failed: ${e.oauthError ?: "?"} ${e.oauthErrorDescription ?: ""}")
} catch (e: TransportException) {
    println(if (e.isTimeout) "timed out" else e.message)
}

// LwsException is sealed, so a when over it is exhaustive.
fun describe(e: LwsException): String = when (e) {
    is HttpException -> "HTTP ${e.status}"
    is AuthenticationException -> "authorization failed"
    is ProtocolException -> "the server broke the protocol"
    is SignatureVerificationException -> "a forged notification"
    is TransportException -> "no response"
}
