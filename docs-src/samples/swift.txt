@@@ conditional
let first = try await client.read(url)
guard let etag = first.etag else { fatalError("the server sent no ETag") }

// Revalidate: 304 Not Modified is a normal result, not an error.
let again = try await client.read(url, options: ReadOptions(ifNoneMatch: etag))
if again.notModified { print("cached copy is still fresh") }

// Optimistic concurrency: only overwrite the version you read.
do {
    _ = try await client.update(url, body: Data("new content".utf8), contentType: "text/plain", options: UpdateOptions(ifMatch: etag))
} catch LWSError.preconditionFailed {
    // Someone else changed it (412): re-read, merge, retry.
}

@@@ patch-linkset
let doc = try await client.readLinkset(url)
let rel = "https://example.org/rel/reviewer"         // extension relation types are URIs
let reviewer: JSONValue = ["href": "https://id.example/bob"]

// JSONPointer escapes "/" in the relation URI as "~1".
// "add" on an existing member replaces it, so append with "/-" when the relation exists.
let patch = doc.linkset.targets(rel).isEmpty
    ? JSONPatch().add(JSONPointer.from("linkset", "0", rel), [reviewer])
    : JSONPatch().add(JSONPointer.from("linkset", "0", rel, "-"), reviewer)
_ = try await client.patchLinkset(doc.url, patch: patch, options: UpdateOptions(ifMatch: doc.etag))

@@@ custom-authenticator
import Foundation
import LWS

// Any scheme (API keys, cookies, DPoP…) plugs in through the Authenticator protocol.
struct APIKeyAuthenticator: Authenticator {
    let key: String

    func authorize(_ request: AuthRequest) async throws -> AuthRequest {
        guard request.url.host == "storage.example" else { return request }
        var r = request
        r.headers.set("X-Api-Key", key)
        return r
    }

    // Nothing to refresh: the 401 surfaces as LWSError.unauthorized.
    func handleChallenge(_ request: AuthRequest, response: AuthResponse) async throws -> Bool { false }
}

let client = LWSClient(authenticator: APIKeyAuthenticator(key: "my-api-key"))

@@@ webhook
import Foundation
import LWS

let client = LWSClient()                             // fetches storage descriptions
let verifier = WebhookVerifier(options: WebhookVerifierOptions(
    client: client,
    trustedStorages: [URL(string: "https://storage.example/")!]   // only this storage's keys
))
let inbox = URL(string: "https://my-app.example/hooks/lws")!      // the registered (public) URL

// In your HTTP server's handler (Vapor, Hummingbird, SwiftNIO, …), with the request's parts:
func handle(method: String, headers: [(String, String)], body: Data) async -> Int {
    do {
        let verified = try await verifier.verify(method: method, url: inbox, headers: HTTPHeaders(headers), body: body)
        for a in verified.notification.activities { print(a.types, a.object.id, a.published as Any) }
        return 204
    } catch LWSError.signatureVerification(let reason) {
        print("rejected: \(reason)")
        return 401
    } catch {
        return 500
    }
}

@@@ errors

    _ = try await client.read(url)
} catch LWSError.notFound {
    print("no such resource")
} catch LWSError.unauthorized(let e) {
    print("not authenticated", e.challenges)
