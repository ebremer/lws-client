// Any scheme (API keys, cookies, DPoP…) plugs in by implementing lws::Authenticator.
class ApiKeyAuthenticator final : public lws::Authenticator {
public:
    explicit ApiKeyAuthenticator(std::string key) : key_(std::move(key)) {}

    void authorize(lws::HttpRequest& request) override {
        if (request.url.starts_with("https://storage.example/")) request.headers.set("X-Api-Key", key_);
    }

    bool handle_challenge(const lws::HttpRequest&, const lws::HttpResponse&, lws::HttpTransport&) override {
        return false;   // nothing to refresh: the 401 surfaces as lws::UnauthorizedError
    }

private:
    std::string key_;
};

lws::Client make_api_key_client(std::string api_key) {
    lws::ClientOptions options;
    options.authenticator = std::make_shared<ApiKeyAuthenticator>(std::move(api_key));
    return lws::Client(options);
}

#include <iostream>
#include <lws/lws.hpp>

// Call from your HTTP server (Boost.Beast, Drogon, cpp-httplib, …) with the raw request.
int handle_delivery(lws::WebhookVerifier& verifier, std::string_view method,
                    const lws::HttpHeaders& headers, std::string_view body) {
    // The URL you registered (your public URL when behind a proxy), not the Host header.
    constexpr std::string_view inbox = "https://my-app.example/hooks/lws";
    try {
        lws::VerifiedNotification verified = verifier.verify(method, inbox, headers, body);
        for (const lws::Activity& a : verified.notification.activities) {
            std::cout << a.object.id << (a.is_update() ? " updated" : "") << '\n';
        }
        return 204;
    } catch (const lws::SignatureVerificationError& e) {
        std::cerr << "rejected: " << e.what() << '\n';
        return 401;
    }
}
