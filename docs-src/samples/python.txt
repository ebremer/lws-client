import httpx


# Any scheme (API keys, cookies, DPoP…) plugs in by subclassing Authenticator.
class ApiKeyAuthenticator(Authenticator):
    def __init__(self, key: str) -> None:
        self.key = key

    def authorize(self, request: httpx.Request) -> None:
        if request.url.host == "storage.example":
            request.headers["x-api-key"] = self.key

@@@ webhook
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

from lws_client import LwsClient, SignatureVerificationError, WebhookVerifier

INBOX = "https://my-app.example/hooks/lws"   # the registered URL (your public URL behind a proxy)
verifier = WebhookVerifier(client=LwsClient(), trusted_storages=["https://storage.example/"])


class Inbox(BaseHTTPRequestHandler):
    def do_POST(self) -> None:
        body = self.rfile.read(int(self.headers.get("Content-Length", "0")))
        try:
            verified = verifier.verify("POST", INBOX, list(self.headers.items()), body)
        except SignatureVerificationError as exc:
            print("rejected:", exc)
            self.send_response(401)
        else:
            for activity in verified.notification.activities:
                print(activity.types, activity.object.id, activity.published)
            self.send_response(204)
        self.end_headers()


ThreadingHTTPServer(("0.0.0.0", 8080), Inbox).serve_forever()
# ASGI apps: await AsyncWebhookVerifier(...).verify_asgi(scope, receive, INBOX)

@@@ async-usage
import asyncio

from lws_client import AsyncLwsClient, SelfSignedCredentials, SigningKey, TokenExchangeAuthenticator


async def main() -> None:
    credentials = SelfSignedCredentials.did_key(SigningKey.generate())
    async with AsyncLwsClient(authenticator=TokenExchangeAuthenticator(credentials)) as client:
        storage = await client.discover_storage("http://localhost:8787/root/")
        root = storage.storage_root()
        # Concurrent requests share a single token exchange.
        pages = await asyncio.gather(*(client.read_container(root) for _ in range(5)))
        print([page.total_items for page in pages])
        async for item in client.list_container(root):
            print(item.id)


asyncio.run(main())
