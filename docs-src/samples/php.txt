@@@ read
$resource = $client->read($url);
$meta = $resource->metadata;
echo $resource->status, ' ', $resource->contentType ?? '', ' ', $resource->etag ?? '', ' ', $meta->lastModified ?? '', "\n";
echo implode(', ', $meta->types), ' ', $meta->isDataResource() ? 'data' : 'container', ' ', $meta->parent ?? '', "\n";

@@@ conditional
$first = $client->read($url);
$etag = $first->etag ?? throw new RuntimeException('the server sent no ETag');

// Revalidate: 304 Not Modified is a normal result, not an error.
if ($client->read($url, ifNoneMatch: $etag)->notModified) {
    echo "cached copy is still fresh\n";
}

// Optimistic concurrency: only overwrite the version you read.
try {
    $client->update($url, 'new content', 'text/plain', ifMatch: $etag);
} catch (PreconditionFailedException) {
    // Someone else changed it (412): re-read, merge, retry.
}
@@@ range
$start = $client->read($url, range: ByteRange::of(0, 1023));    // first KiB
echo $start->status, ' ', $start->contentRange ?? '', "\n";     // 206 bytes 0-1023/52311

$rest = $client->read($url, range: ByteRange::from(1024));      // to the end
$tail = $client->read($url, range: ByteRange::last(512));       // last 512 bytes
echo strlen($rest->body), ' ', strlen($tail->body), "\n";
@@@ update
// PUT replaces the whole representation.
$client->update($url, Json::encode(['name' => 'Alice', 'age' => 31]), 'application/json', ifMatch: $etag);

// Replace content and user-managed links atomically (Prefer: set-linkset).
$client->update($url, 'v2', 'text/plain',
    links: [new Link('https://creativecommons.org/licenses/by/4.0/', 'license')],
    setLinkset: true,

@@@ patch
// JSON Patch (RFC 6902) is the baseline format every LWS server accepts.
$patch = (new JsonPatch())
    ->test('/version', 3)
    ->replace('/status', 'done')
    ->add('/tags/-', 'urgent')
    ->remove('/draft');
$client->patch($url, $patch, ifMatch: $etag);

// Other formats only when the server advertises them in Accept-Patch.
if (in_array('application/sparql-update', $client->head($url)->acceptPatch, true)) {
    $client->patch($url, 'INSERT DATA { <#me> <http://xmlns.com/foaf/0.1/name> "Alice" }', 'application/sparql-update');
}

@@@ patch-linkset
$doc = $client->readLinkset($url);
$rel = 'https://example.org/rel/reviewer';           // extension relation types are URIs
$reviewer = ['href' => 'https://id.example/bob'];

// JsonPointer escapes "/" in the relation URI as "~1".
// "add" on an existing member replaces it, so append with "/-" when the relation exists.
$patch = $doc->linkset->targets($rel) === []
    ? (new JsonPatch())->add(JsonPointer::fromSegments('linkset', '0', $rel), [$reviewer])
    : (new JsonPatch())->add(JsonPointer::fromSegments('linkset', '0', $rel, '-'), $reviewer);
$client->patchLinkset($doc->url, $patch, ifMatch: $doc->etag);

@@@ custom-authenticator
<?php
require __DIR__ . '/vendor/autoload.php';

use Ebremer\Lws\Auth\Authenticator;
use Ebremer\Lws\Auth\AuthRequest;
use Ebremer\Lws\Auth\AuthResponse;
use Ebremer\Lws\LwsClient;

// Any scheme (API keys, cookies, DPoP…) plugs in through the Authenticator interface.
final class ApiKeyAuthenticator implements Authenticator
{
    public function __construct(#[\SensitiveParameter] private readonly string $key)
    {
    }

    public function authorize(AuthRequest $request): AuthRequest
    {
        if (parse_url($request->url, PHP_URL_HOST) !== 'storage.example') {
            return $request;
        }
        return $request->withHeaders($request->headers->with('X-Api-Key', $this->key));
    }

    // Nothing to refresh: the 401 surfaces as UnauthorizedException.
    public function handleChallenge(AuthRequest $request, AuthResponse $response): bool
    {
        return false;
    }
}

$client = new LwsClient(authenticator: new ApiKeyAuthenticator('my-api-key'));

@@@ webhook
<?php
// hooks/lws.php, the inbox (under PHP-FPM, Apache mod_php, or php -S).
require __DIR__ . '/vendor/autoload.php';

use Ebremer\Lws\Exception\SignatureVerificationException;
use Ebremer\Lws\LwsClient;
use Ebremer\Lws\Notification\WebhookVerifier;

$verifier = new WebhookVerifier(
    client: new LwsClient(),                         // fetches storage descriptions
    trustedStorages: ['https://storage.example/'],   // only this storage's keys
);
try {
    // The registered (public) inbox URL, not the one a proxy forwarded to. With a PSR-7 framework,
    // call $verifier->verifyServerRequest($request, $inbox) instead.
    $verified = $verifier->verifyGlobals('https://my-app.example/hooks/lws');
    foreach ($verified->notification->activities as $a) {
        error_log(implode(',', $a->types) . ' ' . $a->object->id . ' ' . ($a->publishedRaw ?? ''));
    }
    http_response_code(204);
} catch (SignatureVerificationException $e) {
    error_log('rejected: ' . $e->getMessage());
    http_response_code(401);
}

@@@ errors
try {
    $client->read($url);
} catch (NotFoundException) {
    echo "no such resource\n";
} catch (UnauthorizedException $e) {
    echo 'not authenticated: ', count($e->challenges()), " challenge(s)\n";
} catch (HttpException $e) {                         // any other error status
    echo $e->status, ' ', $e->problem->title ?? $e->getMessage(), "\n";
} catch (AuthenticationException $e) {
    echo 'token exchange failed: ', $e->oauthError ?? '?', ' ', $e->oauthErrorDescription ?? '', "\n";
} catch (TransportException $e) {
    echo $e->isTimeout() ? "timed out\n" : $e->getMessage() . "\n";
}
