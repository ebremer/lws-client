@@@ conditional
Resource first = await client.ReadAsync(url);
string etag = first.ETag ?? throw new InvalidOperationException("the server sent no ETag");

// Revalidate: 304 Not Modified is a normal result, not an exception.
Resource again = await client.ReadAsync(url, new ReadOptions { IfNoneMatch = etag });
if (again.NotModified) Console.WriteLine("cached copy is still fresh");

// Optimistic concurrency: only overwrite the version you read.
try
{
    await client.UpdateAsync(url, "new content"u8.ToArray(), "text/plain", new UpdateOptions { IfMatch = etag });
}
catch (PreconditionFailedException)
{
    // Someone else changed it (412): re-read, merge, retry.
}

// JSON Patch (RFC 6902) is the baseline format every LWS server accepts. JsonPatch is immutable.

    .Test("/version", 3)
    .Replace("/status", "done")
    .Add("/tags/-", "urgent")
    .Remove("/draft");

LinksetDocument doc = await client.ReadLinksetAsync(url);
string rel = "https://example.org/rel/reviewer";             // extension relation types are URIs
var reviewer = new JsonObject { ["href"] = "https://id.example/bob" };

// JsonPointer escapes "/" in the relation URI as "~1".
// "add" on an existing member replaces it, so append with "/-" when the relation exists.
JsonPatch patch = doc.Linkset.GetTargets(rel).Count == 0
    ? new JsonPatch().Add(JsonPointer.Of("linkset", "0", rel), new JsonArray(reviewer))
    : new JsonPatch().Add(JsonPointer.Of("linkset", "0", rel, "-"), reviewer);
await client.PatchLinksetAsync(doc.Url, patch, new UpdateOptions { IfMatch = doc.ETag });

sealed class ApiKeyAuthenticator(string key) : IAuthenticator
{
    public ValueTask AuthorizeAsync(AuthRequest request, CancellationToken cancellationToken = default)
    {
        if (request.Uri.Host == "storage.example") request.SetHeader("X-Api-Key", key);
        return ValueTask.CompletedTask;
    }

    // Nothing to refresh: the 401 surfaces as UnauthorizedException.
    public ValueTask<bool> HandleChallengeAsync(AuthRequest request, AuthResponse response,
        CancellationToken cancellationToken = default) => ValueTask.FromResult(false);
}

@@@ webhook
using System.Net;
using Ebremer.Lws;
using Ebremer.Lws.Notifications;

using var client = new LwsClient();                                      // fetches storage descriptions
using var verifier = new WebhookVerifier(new WebhookVerifierOptions
{
    Client = client,
    TrustedStorages = [new Uri("https://storage.example/")],            // only this storage's keys
});
var inbox = new Uri("https://my-app.example/hooks/lws");                 // the registered (public) URL

using var listener = new HttpListener();                                 // behind your reverse proxy
listener.Prefixes.Add("http://localhost:8080/");
listener.Start();
while (true)
{
    HttpListenerContext context = await listener.GetContextAsync();
    try
    {
        VerifiedNotification verified = await verifier.VerifyAsync(context.Request, inbox);
        foreach (Activity a in verified.Notification.Activities)
            Console.WriteLine($"{string.Join(", ", a.Types)} {a.Object.Id} {a.Published}");
        context.Response.StatusCode = 204;
    }
    catch (SignatureVerificationException)
    {
        context.Response.StatusCode = 401;
    }
    finally
    {
        context.Response.Close();
    }
}
// Other frameworks (ASP.NET Core, …): verifier.VerifyAsync(method, inbox, HeaderMap.From(headers), body)
