Errors

Each client maps HTTP statuses to typed errors that line up with the LWS abstract responses (not permitted, unknown requester, target not found, conflict, …). It also attaches the request, the response headers and any RFC 9457 problem details.

All errors share a root: LwsException (Java, unchecked), LwsError (TypeScript, Python), lws::Error : std::runtime_error (C++), or a single enum Error (Rust). Go uses *HTTPError plus sentinels that work with errors.Is and errors.As. Invalid builder input (a relative IRI in a type query, an access policy without actions) raises the language's ordinary argument error instead: IllegalArgumentException, TypeError, std::invalid_argument, ValueError, a returned error, or Rust's Error::InvalidInput.

Handling errors

try {
    client.read(url);
} catch (NotFoundException e) {
    System.out.println("no such resource");
} catch (UnauthorizedException e) {
    System.out.println("not authenticated " + e.challenges());
} catch (HttpStatusException e) {
    System.out.println(e.status() + " " + e.problem().flatMap(ProblemDetails::title).orElse(e.getMessage()));
} catch (AuthenticationException e) {
    System.out.println("token exchange failed: " + e.error().orElse("?") + " " + e.errorDescription().orElse(""));
}
try {
  await client.read(url);
} catch (e) {
  if (e instanceof NotFoundError) {
    console.log("no such resource");
  } else if (e instanceof UnauthorizedError) {
    console.log("not authenticated", e.challenges);
  } else if (e instanceof HttpError) {
    console.log(e.status, e.problem?.title ?? e.message);
  } else if (e instanceof AuthenticationError) {
    console.log("token exchange failed:", e.error, e.errorDescription);
  } else {
    throw e;
  }
}
try {
    client.read(url);
} catch (const lws::NotFoundError&) {
    std::cout << "no such resource\n";
} catch (const lws::UnauthorizedError& e) {
    std::cout << "not authenticated (" << e.challenges().size() << " challenges)\n";
} catch (const lws::HttpError& e) {
    std::cout << e.status() << ' ' << (e.problem() && e.problem()->title ? *e.problem()->title : e.what()) << '\n';
} catch (const lws::AuthenticationError& e) {
    std::cout << "token exchange failed: " << e.oauth_error().value_or("?") << '\n';
}
match client.read(url).await {
    Ok(resource) => println!("{} bytes", resource.body.len()),
    Err(e) if e.is_not_found() => println!("no such resource"),
    Err(Error::Unauthorized(http)) => println!("not authenticated: {:?}", http.challenges),
    Err(Error::Authentication { error, error_description, .. }) => {
        println!("token exchange failed: {error:?} {error_description:?}")
    }
    Err(e) if e.status().is_some() => {
        println!("{:?} {:?}", e.status(), e.problem().and_then(|p| p.title.clone()))
    }
    Err(e) => return Err(e),
}
_, err := client.Read(ctx, url)
var httpErr *lws.HTTPError
var authErr *lws.AuthenticationError
switch {
case err == nil:
case errors.Is(err, lws.ErrNotFound):
	fmt.Println("no such resource")
case errors.Is(err, lws.ErrUnauthorized) && errors.As(err, &httpErr):
	fmt.Println("not authenticated", httpErr.Challenges)
case errors.As(err, &httpErr):
	fmt.Println(httpErr.StatusCode, httpErr.Problem)
case errors.As(err, &authErr):
	fmt.Println("token exchange failed:", authErr.Code, authErr.Description)
default:
	return err
}
try:
    client.read(url)
except NotFoundError:
    print("no such resource")
except UnauthorizedError as exc:
    print("not authenticated", exc.challenges)
except HttpError as exc:
    print(exc.status, exc.problem.title if exc.problem else exc)
except AuthenticationError as exc:
    print("token exchange failed:", exc.error, exc.error_description)

Problem details

When an error response has an RFC 9457 body (application/problem+json), the error carries it as problem: type, title, status, detail, instance and any extension members. The raw response body is kept too (Python keeps the first 4 KiB).

try {
    client.delete(container);
} catch (ConflictException e) {
    e.problem().ifPresent(p -> System.out.println(p.type() + " " + p.title() + " " + p.detail()
            + " " + p.instance() + " " + p.extensions().get("itemCount")));
}
try {
  await client.delete(container);
} catch (e) {
  if (e instanceof ConflictError && e.problem) {
    const { type, title, detail, instance, extensions } = e.problem;
    console.log(type, title, detail, instance, extensions["itemCount"]);
  }
}
try {
    client.remove(container);
} catch (const lws::ConflictError& e) {
    if (const auto& p = e.problem()) {
        std::cout << p->type.value_or("") << ' ' << p->title.value_or("") << ' ' << p->detail.value_or("") << ' '
                  << p->instance.value_or("") << ' ' << p->extensions.value("itemCount", 0) << '\n';
    }
}
if let Err(e) = client.delete(container).await {
    if let (true, Some(p)) = (e.is_conflict(), e.problem()) {
        println!(
            "{:?} {:?} {:?} {:?} {:?}",
            p.problem_type, p.title, p.detail, p.instance, p.extensions.get("itemCount")
        );
    }
}
err := client.Delete(ctx, container)
var httpErr *lws.HTTPError
if errors.Is(err, lws.ErrConflict) && errors.As(err, &httpErr) && httpErr.Problem != nil {
	p := httpErr.Problem
	fmt.Println(p.Type, p.Title, p.Detail, p.Instance, string(p.Extensions["itemCount"]))
}
try:
    client.delete(container)
except ConflictError as exc:
    if exc.problem is not None:
        p = exc.problem
        print(p.type, p.title, p.detail, p.instance, p.extensions.get("itemCount"))

304 and 206 are not errors A conditional read that returns 304 Not Modified yields a normal result flagged notModified, and 206 Partial Content is a normal success.
