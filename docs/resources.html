Reading & writing resources

LWS maps its four abstract operations (create, read, update, delete) onto plain HTTP. The clients add typed metadata, automatic URL resolution and error mapping, so you work with resources rather than raw requests.

About the samples Fragments assume client (an authenticated client), container (a container URL), url (a resource URL) and etag are in scope. See Getting started for a complete program.

Create

Resources are always created inside an existing container with POST. The server assigns the final URL and returns it in Location, which the clients surface (absolute) as the create result's location. A container is created by sending Link: <https://www.w3.org/ns/lws#Container>; rel="type" with an empty body, which createContainer does for you.

// A container (POST with Link: <https://www.w3.org/ns/lws#Container>; rel="type")
URI notes = client.createContainer(container, CreateOptions.slug("notes")).location();

// Text, JSON or any bytes. The slug is only a hint: the server picks the final URL.
CreateResult list = client.create(notes, Body.of("milk\neggs\n"), "text/plain", CreateOptions.slug("shopping.txt"));
CreateResult todo = client.createJson(notes, Map.of("task", "write docs"), CreateOptions.builder()
        .slug("todo.json")
        .type("https://schema.org/Action")                                // extra rel="type"
        .link(Link.of("https://example.org/shapes/todo", "describedby"))  // user metadata
        .build());
CreateResult photo = client.create(notes, Body.ofFile(Path.of("photo.png")), "image/png",
        CreateOptions.slug("photo.png"));

System.out.println(list.location() + " " + todo.metadata().linkset() + " " + photo.metadata().parent());

OptionWire effect
slugSlug header: a hint for the new name (percent-encoded). The server may change it to keep names unique.
typesExtra Link: <iri>; rel="type" headers. Type index and search use them.
linksAny other user-managed metadata, sent as Link headers and stored in the resource's linkset.

POST is not idempotent The clients never retry a POST automatically. The one exception is the single retry after a 401 authentication challenge, when nothing was created.

Read

Streaming large bodies

Where the platform supports it, the clients can hand you the body as a stream instead of buffering it. Streamed uploads cannot be replayed after a 401, so the JS, Python, Go and C# clients first send a HEAD to obtain a token. Java bodies are always replayable. C++ bodies are buffered strings, so use range requests to process large resources in pieces.

try (ResourceStream stream = client.readStream(url)) {    // nothing is buffered
    long received = stream.body().transferTo(OutputStream.nullOutputStream());
    System.out.println("received " + received + " bytes");
}

// Uploads: every Body is replayable (files are re-read), so the 401 retry always works.
client.create(container, Body.ofFile(Path.of("big.bin")), "application/octet-stream",
        CreateOptions.slug("big.bin"));
const resource = await client.read(url);
const reader = resource.body!.getReader();   // ReadableStream<Uint8Array>, nothing buffered
let received = 0;
for (;;) {
  const { done, value } = await reader.read();
  if (done) break;
  received += value.byteLength;
}
console.log(`received ${received} bytes`);

// Upload a stream: the client obtains a token first (HEAD) because a stream cannot be replayed.
const upload = new Blob(["large file contents"]).stream();
await client.create(container, upload, "application/octet-stream", { slug: "big.bin" });
// Bodies are buffered std::strings. Process large resources piecewise with range requests:
const std::uint64_t total = client.head(url).content_length.value_or(0);
constexpr std::uint64_t chunk = 1 << 20;
for (std::uint64_t offset = 0; offset < total; offset += chunk) {
    auto part = client.read(url, {.range = lws::byte_range(offset, std::min(total, offset + chunk) - 1)});
    std::cout << "received " << part.body.size() << " bytes\n";
}
let mut resource = client.read(url).send_streaming().await?;   // nothing is buffered
let mut received = 0;
while let Some(chunk) = resource.chunk().await? {
    received += chunk.len();
}
println!("received {received} bytes");
s, err := client.ReadStream(ctx, url) // nothing is buffered
if err != nil {
	return err
}
defer s.Body.Close()
n, err := io.Copy(io.Discard, s.Body)
if err != nil {
	return err
}
fmt.Println("received", n, "bytes")

// Uploads stream from any io.Reader. Without a cached token the client sends a HEAD first,
// because a stream cannot be replayed after a 401.
f, err := os.Open("big.bin")
if err != nil {
	return err
}
defer f.Close()
_, err = client.Create(ctx, container, f, "application/octet-stream", lws.Slug("big.bin"))
return err
from collections.abc import Iterator

received = 0
with client.stream(url) as response:      # nothing is buffered
    for chunk in response.iter_bytes():
        received += len(chunk)
print(f"received {received} bytes")


# Upload from an iterator: the client obtains a token first (HEAD), because a
# generator cannot be replayed after a 401.
def chunks() -> Iterator[bytes]:
    yield b"large file "
    yield b"contents"


client.create(container, chunks(), "application/octet-stream", slug="big.bin")
await using (ResourceStream stream = await client.ReadStreamAsync(url))      // nothing is buffered
await using (FileStream copy = File.Create("copy.bin"))
{
    await stream.Content.CopyToAsync(copy);
    Console.WriteLine($"received {copy.Length} bytes");
}

// A Stream body is sent once: the client obtains a token first (HEAD), because it cannot replay it after a 401.
await using FileStream upload = File.OpenRead("big.bin");
await client.CreateAsync(container, upload, "application/octet-stream", new CreateOptions { Slug = "big.bin" });

Conditional requests

Every LWS response to GET/HEAD carries an ETag. Echo it back verbatim, quotes included:

  • ifNoneMatch on a read revalidates a cached copy. A 304 is returned as a normal result flagged notModified, not as an error.
  • ifMatch on update, patch or delete prevents lost updates. A mismatch raises the precondition failed error (412).
Resource first = client.read(url);
String etag = first.etag().orElseThrow();

// Revalidate: 304 Not Modified is a normal result, not an exception.
Resource again = client.read(url, ReadOptions.ifNoneMatch(etag));
if (again.notModified()) System.out.println("cached copy is still fresh");

// Optimistic concurrency: only overwrite the version you read.
try {
    client.update(url, Body.of("new content"), "text/plain", UpdateOptions.ifMatch(etag));
} catch (PreconditionFailedException e) {
    // Someone else changed it (412): re-read, merge, retry.
}
const first = await client.read(url);

// Revalidate: 304 Not Modified is a normal result, not an error.
const again = await client.read(url, { ifNoneMatch: first.etag! });
if (again.notModified) console.log("cached copy is still fresh");

// Optimistic concurrency: only overwrite the version you read.
try {
  await client.update(url, "new content", "text/plain", { ifMatch: first.etag! });
} catch (e) {
  if (!(e instanceof PreconditionFailedError)) throw e;
  // Someone else changed it (412): re-read, merge, retry.
}
lws::Resource first = client.read(url);

// Revalidate: 304 Not Modified is a normal result, not an exception.
lws::Resource again = client.read(url, {.if_none_match = first.etag});
if (again.not_modified) std::cout << "cached copy is still fresh\n";

// Optimistic concurrency: only overwrite the version you read.
try {
    client.update(url, "new content", "text/plain", {.if_match = first.etag});
} catch (const lws::PreconditionFailedError&) {
    // Someone else changed it (412): re-read, merge, retry.
}
let first = client.read(url).await?;
let etag = first.metadata.etag.clone().unwrap_or_default();

// Revalidate: 304 Not Modified is a normal result, not an error.
let again = client.read(url).if_none_match(&etag).await?;
if again.not_modified {
    println!("cached copy is still fresh");
}

// Optimistic concurrency: only overwrite the version you read.
match client.update(url, "new content", "text/plain").if_match(&etag).await {
    Ok(_) => {}
    Err(e) if e.is_precondition_failed() => { /* 412: re-read, merge, retry */ }
    Err(e) => return Err(e),
}
first, err := client.Read(ctx, url)
if err != nil {
	return err
}

// Revalidate: 304 Not Modified is a normal result, not an error.
again, err := client.Read(ctx, url, lws.IfNoneMatch(first.ETag))
if err != nil {
	return err
}
if again.NotModified {
	fmt.Println("cached copy is still fresh")
}

// Optimistic concurrency: only overwrite the version you read.
_, err = client.Update(ctx, url, strings.NewReader("new content"), "text/plain", lws.IfMatch(first.ETag))
if errors.Is(err, lws.ErrPreconditionFailed) {
	// Someone else changed it (412): re-read, merge, retry.
} else if err != nil {
	return err
}
first = client.read(url)

# Revalidate: 304 Not Modified is a normal result, not an error.
again = client.read(url, if_none_match=first.etag)
if again.not_modified:
    print("cached copy is still fresh")

# Optimistic concurrency: only overwrite the version you read.
try:
    client.update(url, "new content", "text/plain", if_match=first.etag)
except PreconditionFailedError:
    pass   # someone else changed it (412): re-read, merge, retry
Resource first = await client.ReadAsync(url);
string etag = first.ETag ?? throw new InvalidOperationException("the server sent no ETag");

// Revalidate: 304 Not Modified is a normal result, not an exception.
Resource again = await client.ReadAsync(url, new ReadOptions { IfNoneMatch = etag });
if (again.NotModified) Console.WriteLine("cached copy is still fresh");

// Optimistic concurrency: only overwrite the version you read.
try
{
    await client.UpdateAsync(url, "new content"u8.ToArray(), "text/plain", new UpdateOptions { IfMatch = etag });
}
catch (PreconditionFailedException)
{
    // Someone else changed it (412): re-read, merge, retry.
}

Byte ranges

Servers must support range requests for data resources. A 206 Partial Content is a normal success, and the clients expose Content-Range.

Resource start = client.read(url, ReadOptions.builder().range(0, 1023L).build());   // first KiB
System.out.println(start.status() + " " + start.contentRange().orElse(""));         // 206 bytes 0-1023/52311

Resource rest = client.read(url, ReadOptions.builder().range(1024, null).build());   // to the end
Resource tail = client.read(url, ReadOptions.builder().suffixRange(512).build());    // last 512 bytes
System.out.println(rest.body().length + " " + tail.body().length);
const start = await client.read(url, { range: { start: 0, end: 1023 } });   // first KiB
console.log(start.partial, start.contentRange);                            // true "bytes 0-1023/52311"

const tail = await client.read(url, { range: { suffix: 512 } });           // last 512 bytes
const rest = await client.read(url, { range: "bytes=1024-" });             // raw Range value
console.log((await tail.bytes()).byteLength, rest.status);
auto start = client.read(url, {.range = lws::byte_range(0, 1023)});    // first KiB
std::cout << start.status << ' ' << start.content_range.value_or("") << '\n';   // 206 bytes 0-1023/52311

auto rest = client.read(url, {.range = lws::byte_range(1024)});         // to the end
auto tail = client.read(url, {.range = "bytes=-512"});                  // raw Range value: last 512 bytes
std::cout << rest.body.size() << ' ' << tail.body.size() << '\n';
let start = client.read(url).range(0, Some(1023)).await?;   // first KiB
println!("{} {:?}", start.is_partial(), start.metadata.content_range());   // true "bytes 0-1023/52311"

let rest = client.read(url).range(1024, None).await?;       // to the end
println!("{}", rest.body.len());
start, err := client.Read(ctx, url, lws.ByteRange(0, 1023)) // first KiB
if err != nil {
	return err
}
fmt.Println(start.StatusCode, start.ContentRange) // 206 "bytes 0-1023/52311"

rest, err := client.Read(ctx, url, lws.ByteRange(1024, -1)) // from byte 1024 to the end
if err != nil {
	return err
}
tail, err := client.Read(ctx, url, lws.ByteRange(-512, 0)) // the last 512 bytes
if err != nil {
	return err
}
fmt.Println(len(rest.Body), len(tail.Body))
start = client.read(url, range=(0, 1023))        # first KiB
print(start.is_partial, start.content_range)     # True "bytes 0-1023/52311"

rest = client.read(url, range=(1024, None))      # from byte 1024 to the end
tail = client.read(url, range="bytes=-512")      # raw Range value: the last 512 bytes
print(len(rest.content), len(tail.content))
Resource start = await client.ReadAsync(url, new ReadOptions { Range = new RangeHeaderValue(0, 1023) });   // first KiB
Console.WriteLine($"{start.Status} {start.ContentRange}");                                                  // 206 bytes 0-1023/52311

Resource rest = await client.ReadAsync(url, new ReadOptions { Range = new RangeHeaderValue(1024, null) });  // to the end
Resource tail = await client.ReadAsync(url, new ReadOptions { Range = new RangeHeaderValue(null, 512) });   // last 512 bytes
Console.WriteLine($"{rest.Body.Length} {tail.Body.Length}");

Replace (PUT)

update replaces the whole representation, and the media type should match the existing one. Content updates leave the linkset untouched unless you ask for a combined update with setLinkset. That option sends Prefer: set-linkset plus your Link headers, and the server applies both atomically. Servers that don't support it ignore the preference or answer 501.

// PUT replaces the whole representation.
client.update(url, Body.ofJson(Map.of("name", "Alice", "age", 31)), "application/json", UpdateOptions.ifMatch(etag));

// Replace content and user-managed links atomically (Prefer: set-linkset).
client.update(url, Body.of("v2"), "text/plain", UpdateOptions.builder()
        .setLinkset(true)
        .link(Link.of("https://creativecommons.org/licenses/by/4.0/", "license"))
        .build());
// PUT replaces the whole representation.
await client.update(url, JSON.stringify({ name: "Alice", age: 31 }), "application/json", { ifMatch: etag });
await client.updateJson(url, { name: "Alice", age: 32 });

// Replace content and user-managed links atomically (Prefer: set-linkset).
await client.update(url, "v2", "text/plain", {
  setLinkset: true,
  links: [{ href: "https://creativecommons.org/licenses/by/4.0/", rel: "license" }],
});
// PUT replaces the whole representation.
client.update(url, R"({"name":"Alice","age":31})", "application/json", {.if_match = etag});

// Replace content and user-managed links atomically (Prefer: set-linkset).
lws::UpdateOptions options;
options.set_linkset = true;

Partial updates (PATCH)

JSON Patch (RFC 6902, application/json-patch+json) is the baseline format every LWS server supports. It replaced JSON Merge Patch in the 2026-10-05 draft. Every client includes a JsonPatch builder with add, remove, replace, move, copy and test. A server may accept other formats, such as SPARQL Update for RDF, but only send them when Accept-Patch lists them. Otherwise expect 415.

// JSON Patch (RFC 6902) is the baseline format every LWS server accepts.
client.patch(url, JsonPatch.builder()
        .test("/version", 3)
        .replace("/status", "done")
        .add("/tags/-", "urgent")
        .remove("/draft")
        .build(), UpdateOptions.ifMatch(etag));

// Other formats only when the server advertises them in Accept-Patch.
if (client.head(url).acceptPatch().contains("application/sparql-update")) {
    client.patch(url, Body.of("INSERT DATA { <#me> <http://xmlns.com/foaf/0.1/name> \"Alice\" }"),
            "application/sparql-update", UpdateOptions.DEFAULT);
}

  { ifMatch: etag },
);

Delete

Deleting a resource also removes its linkset and its entry in the parent container. A non-empty container can only be deleted recursively, which sends Depth: infinity. Without it the server answers 409 Conflict.

client.delete(url, DeleteOptions.ifMatch(etag));

try {
    client.delete(container);                             // 409 Conflict when not empty
} catch (ConflictException e) {
    client.delete(container, DeleteOptions.recursive());  // Depth: infinity
}
await client.delete(url, { ifMatch: etag });

try {
  await client.delete(container);                       // 409 Conflict when not empty
} catch (e) {
  if (!(e instanceof ConflictError)) throw e;
  await client.delete(container, { recursive: true });  // Depth: infinity
}
client.remove(url, {.if_match = etag});           // `delete` is a C++ keyword

try {
    client.remove(container);                      // 409 Conflict when not empty
} catch (const lws::ConflictError&) {
    client.remove(container, {.recursive = true}); // Depth: infinity
}
client.delete(url).if_match(etag).await?;

match client.delete(container).await {
    // 409 Conflict when not empty: delete recursively (Depth: infinity)
    Err(e) if e.is_conflict() => client.delete(container).recursive(true).await?,
    other => other?,
}
if err := client.Delete(ctx, url, lws.IfMatch(etag)); err != nil {
	return err
}

err := client.Delete(ctx, container) // 409 Conflict when not empty
if errors.Is(err, lws.ErrConflict) {
	err = client.Delete(ctx, container, lws.Recursive()) // Depth: infinity
}
return err
client.delete(url, if_match=etag)

try:
    client.delete(container)                  # 409 Conflict when not empty
except ConflictError:
    client.delete(container, recursive=True)  # Depth: infinity
await client.DeleteAsync(url, new DeleteOptions { IfMatch = etag });

try
{
    await client.DeleteAsync(container);                                          // 409 Conflict when not empty
}
catch (ConflictException)
{
    await client.DeleteAsync(container, new DeleteOptions { Recursive = true });  // Depth: infinity
}
